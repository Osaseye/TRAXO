        {lockedSignal && !loading && (
          <div className="absolute z-30 left-3 top-16 sm:left-4 sm:top-20 w-[340px] sm:w-[360px] max-w-[calc(100vw-0.9rem)] rounded-2xl bg-[#0d1117] border border-white/[0.1] shadow-2xl p-4">
            <div className="flex items-center justify-between mb-4">
              <div className="flex items-center gap-2">
                <span
                  className={cn(
                    'px-2 py-0.5 rounded text-[10px] font-bold uppercase tracking-wide',
                    lockedSignal.direction === 'BUY'
                      ? 'bg-emerald-500/10 text-emerald-500'
                      : 'bg-rose-500/10 text-rose-500'
                  )}
                >
                  {lockedSignal.direction}
                </span>
                <span className="text-[13px] font-bold text-white">{symbol}</span>
                <span className="text-[10px] font-medium text-slate-300 px-1.5 py-0.5 rounded bg-white/[0.05]">
                  {lockedSignal.strategyLabel}
                </span>
              </div>
              <button
                onClick={() => {
                  setLockedSignalId(null)
                  setHoveredSignalId(null)
                }}
                className="px-2 py-0.5 rounded border border-white/[0.1] text-[10px] font-medium text-slate-400 hover:text-white hover:bg-white/[0.05] transition-colors"
              >
                Close
              </button>
            </div>

            <div className="grid grid-cols-4 gap-2 mb-4">
              {[
                ['ENTRY', lockedSignal.entry, 'text-white'],
                ['SL', lockedSignal.sl, 'text-rose-500'],
                ['TP', lockedSignal.tp, 'text-emerald-500'],
                ['R:R', lockedSignal.rr, 'text-blue-500'],
              ].map(([k, v, color]) => (
                <div key={String(k)}>
                  <span className="block text-[9px] font-semibold text-slate-500 uppercase mb-1">{k}</span>
                  <span className={cn('text-[11px] font-bold', color)}>
                    {typeof v === 'number' && String(k) !== 'R:R' ? formatPrice(v) : typeof v === 'number' ? v.toFixed(1) + 'x' : String(v)}
                  </span>
                </div>
              ))}
            </div>

            {lockedMoneyContext && (
              <div className="flex items-center justify-between text-[11px] font-medium mb-4 pb-4 border-b border-white/[0.05]">
                <div className="flex gap-4">
                  <span className="text-slate-400">Position: <span className="text-white">{fmtUSD(lockedMoneyContext.suggestedPosition)}</span></span>
                  <span className="text-slate-400">Risk: <span className="text-rose-500">{fmtUSD(lockedMoneyContext.maxLossAtSL)}</span></span>
                </div>
                <span className="text-[10px] font-bold text-blue-400 px-1.5 py-0.5 bg-blue-500/10 rounded">
                  {lockedSignal.confidence}% Conf.
                </span>
              </div>
            )}

            <div className="mb-4">
              <div className="text-[9px] font-bold text-slate-500 uppercase tracking-wider mb-2">
                Pattern Analysis
              </div>
              <div className="space-y-1">
                {lockedSignal.reason.map((line, i) => (
                  <p key={i} className="text-[11px] text-slate-300">• {line}</p>
                ))}
              </div>
            </div>

            <div className="flex items-center justify-between pt-4 border-t border-white/[0.05]">
              <span className="text-[11px] text-slate-400">Did you take this trade?</span>
              <div className="flex gap-2">
                <button
                  onClick={() => saveLockedFeedback(true)}
                  className="px-3 py-1 rounded bg-blue-600 hover:bg-blue-700 text-white text-[11px] font-medium transition-colors"
                >
                  Took It
                </button>
                <button
                  onClick={() => saveLockedFeedback(false)}
                  className="px-3 py-1 rounded border border-white/[0.1] text-slate-300 hover:bg-white/[0.05] text-[11px] font-medium transition-colors"
                >
                  Skipped
                </button>
              </div>
            </div>

            {lockedJournal?.taken && lockedSuggestionKey && (
              <div className="pt-3 mt-3 border-t border-white/[0.05]">
                <div className="flex items-center gap-1.5">
                  <button onClick={() => setJournalOutcome(lockedSuggestionKey, 'win')} className={cn("flex-1 h-7 rounded border text-[10px] font-medium transition-colors", lockedJournal.outcome === 'win' ? 'border-emerald-500 bg-emerald-500/20 text-emerald-400' : 'border-white/[0.1] text-slate-400 hover:bg-white/[0.05]')}>Win</button>
                  <button onClick={() => setJournalOutcome(lockedSuggestionKey, 'loss')} className={cn("flex-1 h-7 rounded border text-[10px] font-medium transition-colors", lockedJournal.outcome === 'loss' ? 'border-rose-500 bg-rose-500/20 text-rose-400' : 'border-white/[0.1] text-slate-400 hover:bg-white/[0.05]')}>Loss</button>
                  <button onClick={() => setJournalOutcome(lockedSuggestionKey, 'breakeven')} className={cn("flex-1 h-7 rounded border text-[10px] font-medium transition-colors", lockedJournal.outcome === 'breakeven' ? 'border-slate-400 bg-slate-500/20 text-slate-300' : 'border-white/[0.1] text-slate-400 hover:bg-white/[0.05]')}>BE</button>
                </div>
              </div>
            )}
          </div>
        )}
